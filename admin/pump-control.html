One-time SQL (user-created coins registry)

Run once in Supabase so client-created coins appear here automatically:

CREATE TABLE IF NOT EXISTS pump_coins ( id text PRIMARY KEY, name text, symbol text, emoji text, price numeric, base numeric, created_by uuid, created_at timestamptz DEFAULT now() ); ALTER TABLE pump_coins ENABLE ROW LEVEL SECURITY; DROP POLICY IF EXISTS "pump_coins_read" ON pump_coins; CREATE POLICY "pump_coins_read" ON pump_coins FOR SELECT TO anon, authenticated USING (true); DROP POLICY IF EXISTS "pump_coins_insert" ON pump_coins; CREATE POLICY "pump_coins_insert" ON pump_coins FOR INSERT TO authenticated WITH CHECK (true); DROP POLICY IF EXISTS "pump_coins_update" ON pump_coins; CREATE POLICY "pump_coins_update" ON pump_coins FOR UPDATE TO authenticated USING (true); ALTER TABLE profiles ADD COLUMN IF NOT EXISTS pump_control jsonb DEFAULT NULL;
