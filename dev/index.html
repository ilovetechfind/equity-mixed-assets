Password & database setup (Supabase)

Dev password lives in table dev_settings (column password). Change it only in Supabase Table Editor. Default is 123456. Run this once if the table is missing:

CREATE TABLE IF NOT EXISTS dev_settings ( id int PRIMARY KEY DEFAULT 1 CHECK (id = 1), password text NOT NULL DEFAULT '123456' ); INSERT INTO dev_settings (id, password) VALUES (1, '123456') ON CONFLICT (id) DO NOTHING; -- Allow the app to read the password (needed for the lock screen) ALTER TABLE dev_settings ENABLE ROW LEVEL SECURITY; DROP POLICY IF EXISTS "dev_settings_read" ON dev_settings; CREATE POLICY "dev_settings_read" ON dev_settings FOR SELECT TO anon, authenticated USING (true); -- Optional profile columns for admin access + avatars ALTER TABLE profiles ADD COLUMN IF NOT EXISTS admin_access boolean DEFAULT NULL, ADD COLUMN IF NOT EXISTS admin_status text DEFAULT NULL, ADD COLUMN IF NOT EXISTS role text DEFAULT 'client', ADD COLUMN IF NOT EXISTS avatar_url text DEFAULT NULL;
